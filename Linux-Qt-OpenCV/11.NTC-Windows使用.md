### mainwindow.cpp
NTC 状态调试菜单，发送 `NTC_MSG_STATUS`
```cpp
//NTC状态调试
connect(ui->actionNTC, &QAction::triggered,[=](){
NTCData  ntcMsgSent;
memset(&ntcMsgSent, 0, sizeof(ntcMsgSent));
ntcMsgSent.msgtype  =  NTC_MSG_STATUS;
ntcMsgSent.clientid  =  11;
StateInfo stateInfo;
memset(&stateInfo, 0, sizeof(StateInfo));
stateInfo.fSpeed  =  120;
stateInfo.fLength=  100;

//相机状态灯
QString sState =  nullptr;
for (int  i  =  0; i  <  p_Mainwindow->m_iCamCount; i++)
{
sState+=QString::number(1);
}
strcpy_s(stateInfo.CamState, sState.toLatin1().data());
//Led灯
for(int  i=  0;i  <  MaxCamNum; i++)
{
stateInfo.CamLed[i] =  100;
}

//控制台为自动寻边
if(p_Mainwindow->m_iMeasureWidth  ==  1)
{
stateInfo.CoilEdge  =  p_Mainwindow->m_fEdge;
stateInfo.DetectWidth  =  p_Mainwindow->m_fWidth;//检测幅宽
stateInfo.ProductWidth  =  p_Mainwindow->m_fProduct_Width;//产品幅宽
}

//4组相机的边缘坐标值
for (int  i=0; i<p_Mainwindow->m_iCamCount; i++)
{
if (i  ==  0) stateInfo.GroupEdge[0] =  p_Mainwindow->m_fX_Edge_Cam1;
else  if (i  ==  p_Mainwindow->m_i2ndStartID  -  11) stateInfo.GroupEdge[1] =  p_Mainwindow->m_fX_Edge_Cam2;
else  if (i  ==  p_Mainwindow->m_i3rdStartID  -  11) stateInfo.GroupEdge[2] =  p_Mainwindow->m_fX_Edge_Cam3;
else  if (i  ==  p_Mainwindow->m_i4thStartID  -  11) stateInfo.GroupEdge[3] =  p_Mainwindow->m_fX_Edge_Cam4;
}
memcpy(ntcMsgSent.pData,&stateInfo,sizeof(StateInfo));
widget_tab_offline[0]->m_mutex_NTC.lock();

widget_tab_offline[0]->client.ntcSendMessageToServer((char  *)&ntcMsgSent, sizeof(NTCData));

qDebug() <<"debug---LED Color"<<sState<<"---fSpeed:"  <<  stateInfo.fSpeed  <<  "fLength:"  <<  stateInfo.fLength;

widget_tab_offline[0]->m_mutex_NTC.unlock();
```
